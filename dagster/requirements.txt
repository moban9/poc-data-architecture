# behaviour must not change under us between rebuilds.

#   dagster-postgres 0.29.28 (matches dagster 1.13.x)
#   dbt-trino 1.11.0     (same resolution as the unpinned `dbt` image — pins dbt-core>=1.11.2)

# its declared dependency is psycopg2-binary — install BOTH or instance startup
# dies with "No module named 'psycopg'".
psycopg[binary]==3.3.8
psycopg2-binary==2.9.10
dbt-trino==1.11.0
